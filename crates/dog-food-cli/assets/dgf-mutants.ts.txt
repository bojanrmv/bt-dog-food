

// Added by dog-food (dgf mutate --run) in the mutation sandbox only.
// DGF_MUTANT: one active mutant · DGF_PLAN: "test<TAB>id" lines · DGF_REACH: folder for reach records
function __dgfIs(id: number): boolean {
  const g: any = globalThis;
  const s: any = g.__dgfState || (g.__dgfState = __dgfLoad(g));
  const e: any = g.expect || g[Symbol.for("expect-global")];
  const state: any = e && typeof e.getState === "function" ? e.getState() : undefined;
  const test: string = (state && state.currentTestName) || "";
  const key = test + "\t" + id;
  if (s.reach && !s.seen.has(key)) {
    s.seen.add(key);
    s.fs.appendFileSync(s.reach, key + "\n");
  }
  if (s.plan) {
    return s.plan.get(test) === id;
  }
  return s.active === id;
}

function __dgfLoad(g: any): any {
  const env: any = (g.process && g.process.env) || {};
  const fs: any = g.process && g.process.getBuiltinModule ? g.process.getBuiltinModule("fs") : undefined;
  let plan: Map<string, number> | undefined;
  if (env.DGF_PLAN && fs) {
    plan = new Map<string, number>();
    for (const line of String(fs.readFileSync(env.DGF_PLAN, "utf8")).split("\n")) {
      const tab = line.lastIndexOf("\t");
      if (tab > 0) plan.set(line.slice(0, tab), Number(line.slice(tab + 1)));
    }
  }
  const reach = env.DGF_REACH && fs ? env.DGF_REACH + "/reach-" + Math.random().toString(36).slice(2) + ".txt" : undefined;
  const active = env.DGF_MUTANT === undefined ? -1 : Number(env.DGF_MUTANT);
  return { active, plan, reach, seen: new Set<string>(), fs };
}
